cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(vyanhq-draw VERSION 1.6.0)
option(ENABLE_FRONTEND_API "Use obs-frontend-api" ON)
option(ENABLE_QT "Use Qt" ON)
include(compilerconfig)
include(defaults)
include(helpers)
add_library(vyanhq-draw MODULE src/plugin-main.cpp)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 COMPONENTS Core Widgets Network REQUIRED)
target_link_libraries(vyanhq-draw PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network)
target_compile_features(vyanhq-draw PRIVATE cxx_std_20)
set_target_properties(vyanhq-draw PROPERTIES AUTOMOC ON AUTOUIC ON AUTORCC ON OUTPUT_NAME "vyanhq-draw")
set_target_properties_plugin(vyanhq-draw PROPERTIES OUTPUT_NAME vyanhq-draw)
install(FILES data/locale/en-US.ini DESTINATION data/vyanhq-draw/locale)
